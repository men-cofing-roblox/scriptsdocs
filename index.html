How make Window

MenHub:Notify("hello world!",10 --[[timer sec]]) -- is Notify gang
	  
local window = MenHub:CreateWindow(mylua)
  
window:Button("Click Me!",function()
MenHub:Notify("you click me!")
end)

window:Toggle("Toggle me!",false --[[the def]],function(value)
MenHub:Notify(tostring(value))
end) -- return function for set back to like false
  
window:Silder("Silde me!",0 --[[the def]],0 --[[the min]],1 --[[the max]],true --[[if allow like 0.64]],function(value)
MenHub:Notify(value)
end) -- return function for set back to like 0

window:Dropdown("Drop me!","option1" --[[the def]],{"option1","option2","option3","option4"} --[[the options]],function(value)
MenHub:Notify(value)
end) -- return function for set back to like option1
